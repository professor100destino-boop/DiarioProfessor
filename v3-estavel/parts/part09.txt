function docxParagraphsFromXml(xmlText){
  const doc=new DOMParser().parseFromString(xmlText,'application/xml');
  const ps=[...doc.getElementsByTagNameNS('*','p')];
  return ps.map(p=>{
    const texts=[...p.getElementsByTagNameNS('*','t')].map(t=>t.textContent||'');
    return texts.join('').replace(/\u00a0/g,' ').replace(/\s+/g,' ').trim();
  }).filter(Boolean);
}

function splitPlanejamentoEmAulas(paragraphs){
  const aulas=[];
  let atual=null;
  let campo='raw';

  const setCampo=(p)=>{
    const n=p.normalize('NFD').replace(/[\u0300-\u036f]/g,'').toLowerCase().trim();
    if(/^habilidades?$/.test(n)) return 'habilidade';
    if(/^objetivos? de conhecimentos?\/conteudos?$/.test(n) || /^objetivos?\/conteudos?$/.test(n) || /^objetivos?$/.test(n)) return 'objetivos';
    if(/^metodologia$/.test(n)) return 'metodologia';
    if(/^avaliacao$/.test(n)) return 'avaliacao';
    return null;
  };

  for(const p0 of paragraphs){
    const p=p0.trim();
    const m=p.match(/^AULA\s+(\d+)\s*[–—-]\s*(.+)$/i) || p.match(/^AULA\s+(\d+)\s*[:\-–—]?\s*(.*)$/i);
    if(m){
      if(atual) aulas.push(atual);
      atual={
        numero:Number(m[1]),
        titulo:(m[2]||'').trim() || `Aula ${m[1]}`,
        habilidade:'',
        objetivos:'',
        metodologia:'',
        avaliacao:'',
        raw:''
      };
      campo='raw';
      continue;
    }
    if(!atual) continue;

    const novoCampo=setCampo(p);
    if(novoCampo){
      campo=novoCampo;
      continue;
    }
    atual[campo]=(atual[campo] ? atual[campo]+'\n' : '') + p;
  }
  if(atual) aulas.push(atual);

  return aulas
    .filter(a=>Number.isFinite(a.numero))
    .sort((a,b)=>a.numero-b.numero);
}

async function lerPlanejamentoDocx(file){
  if(typeof JSZip==='undefined') throw new Error('O leitor de DOCX ainda não terminou de carregar. Aguarde alguns segundos e tente novamente.');
  const zip=await JSZip.loadAsync(await file.arrayBuffer());
  const docFile=zip.file('word/document.xml');
  if(!docFile) throw new Error('Este arquivo não parece ser um DOCX válido.');
  const xml=await docFile.async('text');
  const paragraphs=docxParagraphsFromXml(xml);
  const aulas=splitPlanejamentoEmAulas(paragraphs);
  if(!aulas.length) throw new Error('Não encontrei aulas. Use títulos como “AULA 1 – Título da aula”.');
  return aulas;
}

function renderPlanejamentos(){
  const list=$('#planejamentosList');
  if(!list)return;

  const tid=$('#planTurma')?.value||'';
  const plans=db.planejamentos.filter(p=>!tid || p.turmaId===tid);

  if(!plans.length){
    list.innerHTML='<p class=muted style="margin-top:18px">Nenhum planejamento importado para esta turma.</p>';
    $('#planAulasArea')?.classList.add('hidden');
    return;
  }

  list.innerHTML=plans.map(p=>{
    const t=db.turmas.find(x=>x.id===p.turmaId);
    return `<div class="plan-card">
      <div class="plan-card-row">
        <div>
          <b>${esc(p.nome||p.arquivo||'Planejamento')}</b><br>
          <small>${esc(turmaNomeCompleto(t))} • ${p.aulas.length} aulas</small>
        </div>
        <div class="toolbar">
          <button class="primary ghost" onclick="abrirPlanejamento('${p.id}')">Abrir aulas</button>
          <button class="danger" onclick="excluirPlanejamento('${p.id}')">Excluir</button>
        </div>
      </div>
    </div>`;
  }).join('');

  if(planejamentoAtualId && plans.some(p=>p.id===planejamentoAtualId)){
    abrirPlanejamento(planejamentoAtualId,aulaAtualNumero,false);
  }
}
