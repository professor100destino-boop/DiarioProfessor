function load(){
  try{
    const raw=JSON.parse(localStorage.getItem(KEY)||'{}');
    const data={...emptyState(),...raw};
    if(!Array.isArray(data.escolas)) data.escolas=[];
    if(!data.mediasBimestrais || typeof data.mediasBimestrais!=='object') data.mediasBimestrais={};
    if(!Array.isArray(data.planejamentos)) data.planejamentos=[];
    // Migração suave: se havia uma escola escrita nas configurações antigas, vira cadastro de escola.
    if(data.escolas.length===0 && data.config?.escolaNome){
      const eid=idFallback();
      data.escolas.push({id:eid,nome:data.config.escolaNome,cidade:'',rede:'Estadual'});
      data.turmas=(data.turmas||[]).map(t=>({...t,escolaId:t.escolaId||eid}));
    }
    return data;
  }catch{return emptyState()}
}

function escolaOptions(sel, selected){
  const el=$(sel); if(!el)return;
  const prev=selected??el.value;
  el.innerHTML=db.escolas.map(e=>`<option value="${e.id}">${esc(e.nome)}</option>`).join('');
  if(db.escolas.some(e=>e.id===prev))el.value=prev;
}

}

function calcFreqAluno(alunoId){
  const regs=db.frequencias.filter(f=>f.alunoId===alunoId);
  if(!regs.length) return null;
  const presentes=regs.filter(f=>f.status==='P'||f.status==='J'||f.status==='A').length;
  return (presentes/regs.length)*100;
}
