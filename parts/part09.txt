/* === V6.4 Segurança === */
const SECURITY_KEY='professor_control_security_v1';

function securityLoad(){
  try{return JSON.parse(localStorage.getItem(SECURITY_KEY)||'{}')}catch{return{}}
}
function simpleHash(str){
  // Hash local simples para não guardar a senha em texto puro.
  // Não substitui autenticação de servidor, mas protege o uso local.
  let h1=0x811c9dc5;
  for(let i=0;i<str.length;i++){
    h1 ^= str.charCodeAt(i);
    h1 = Math.imul(h1,0x01000193);
  }
  return (h1>>>0).toString(16);
}
function securityCheckPassword(pass){
  const s=securityLoad();
  return !!s.passwordHash && s.passwordHash===simpleHash(pass);
}
function securitySetPassword(pass){
  localStorage.setItem(SECURITY_KEY,JSON.stringify({passwordHash:simpleHash(pass)}));
}

function initSecurity(){
  const lock=document.getElementById('authLock');
  const title=document.getElementById('authTitle');
  const text=document.getElementById('authText');
  const pass=document.getElementById('authPass');
  const pass2=document.getElementById('authPass2');
  const btn=document.getElementById('authBtn');
  const err=document.getElementById('authError');
  if(!lock||!btn)return;

  const sec=securityLoad();
  const first=!sec.passwordHash;

  if(first){
    title.textContent='Criar senha de acesso';
    text.textContent='Este é o primeiro acesso. Crie uma senha para proteger seus dados.';
    pass.placeholder='Crie uma senha';
    pass2.classList.remove('hidden');
    btn.textContent='Salvar senha e entrar';
  }else{
    title.textContent='Professor Control';
    text.textContent='Digite sua senha para acessar o sistema.';
    pass2.classList.add('hidden');
    btn.textContent='Entrar';
  }

  function submit(){
    err.textContent='';
    const p=pass.value;
    if(first){
      if(p.length<4){err.textContent='A senha deve ter pelo menos 4 caracteres.';return}
      if(p!==pass2.value){err.textContent='As senhas não conferem.';return}
      securitySetPassword(p);
      lock.classList.add('hidden');
      pass.value=''; pass2.value='';
    }else{
      if(!securityCheckPassword(p)){err.textContent='Senha incorreta.';return}
      lock.classList.add('hidden');
      pass.value='';
    }
  }

  btn.onclick=submit;
  pass.addEventListener('keydown',e=>{if(e.key==='Enter')submit()});
  pass2.addEventListener('keydown',e=>{if(e.key==='Enter')submit()});
}

function confirmarExclusaoComSenha(tipo,nome){
  const aviso=`ATENÇÃO!\n\nVocê está prestes a excluir ${tipo}: ${nome}.\nEsta ação pode remover dados vinculados e não poderá ser desfeita.\n\nDeseja continuar?`;
  if(!confirm(aviso)) return false;

  const senha=prompt(`Digite a senha do Professor Control para confirmar a exclusão de ${tipo}:`);
  if(senha===null) return false;

  if(!securityCheckPassword(senha)){
    alert('Senha incorreta. A exclusão foi cancelada.');
    return false;
  }
  return true;
}

document.addEventListener('DOMContentLoaded',initSecurity);
</script>
