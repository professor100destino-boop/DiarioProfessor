(function(){
  try{document.body.classList.remove('menu-open')}catch(e){}
  var pass=document.getElementById('authPass');
  var pass2=document.getElementById('authPass2');
  var main=document.getElementById('authBtn');
  var alt=document.getElementById('authAltBtn');
  var kbBtn=document.getElementById('authKeyboardBtn');
  var box=document.getElementById('authKeyboardBox');
  var display=document.getElementById('authKeyboardDisplay');
  var keys=document.getElementById('authKeyboardKeys');
  var shift=document.getElementById('authKeyboardShift');
  var back=document.getElementById('authKeyboardBack');
  var enter=document.getElementById('authKeyboardEnter');
  var value='';
  var upper=false;

  function submitValue(v){
    if(pass){pass.value=v;pass.dispatchEvent(new Event('input',{bubbles:true}));pass.dispatchEvent(new Event('change',{bubbles:true}))}
    if(main)main.click();
  }
  function render(){
    if(display)display.textContent=value?('•'.repeat(value.length)):'Digite pela tela';
    if(!keys)return;
    keys.innerHTML='';
    var chars='1234567890qwertyuiopasdfghjklçzxcvbnmáéíóúãõ@!#$_-*.';
    Array.from(chars).forEach(function(raw){
      var c=upper?raw.toUpperCase():raw;
      var b=document.createElement('button');
      b.type='button';b.textContent=c;
      b.onclick=function(){value+=c;render()};
      keys.appendChild(b);
    });
  }
  if(alt)alt.onclick=function(){
    var v=window.prompt('Digite a mesma senha do Professor Control:');
    if(v===null)return;
    submitValue(v);
  };
  if(kbBtn)kbBtn.onclick=function(){
    if(!box)return;
    box.classList.toggle('hidden');
    render();
  };
  if(shift)shift.onclick=function(){upper=!upper;render()};
  if(back)back.onclick=function(){value=value.slice(0,-1);render()};
  if(enter)enter.onclick=function(){submitValue(value)};
  if(pass){
    pass.disabled=false;pass.readOnly=false;
    pass.addEventListener('pointerdown',function(){try{pass.focus()}catch(e){}},{passive:true});
    pass.addEventListener('click',function(){try{pass.focus()}catch(e){}});
  }
  setTimeout(function(){
    if(pass2&&!pass2.classList.contains('hidden')){
      if(alt)alt.style.display='none';
      if(kbBtn)kbBtn.style.display='none';
      if(box)box.classList.add('hidden');
    }
  },900);
  render();
})();
