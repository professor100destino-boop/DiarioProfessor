<script>
(function(){
  const status=document.getElementById('offlineStatus');

  function setStatus(text,cls){
    if(!status)return;
    status.textContent=text;
    status.className='offline-status '+cls;
  }

  function updateNetworkBadge(){
    if(!navigator.onLine){
      setStatus('✓ Funcionando offline','offline');
    }
  }

  if(!('serviceWorker' in navigator)){
    setStatus('Navegador sem suporte offline','error');
    return;
  }

  if(location.protocol!=='https:' && location.hostname!=='localhost'){
    setStatus('Publique/instale por HTTPS para ativar offline','error');
    return;
  }

  setStatus('Preparando uso 100% offline…','loading');

  window.addEventListener('load', async ()=>{
    try{
      const reg=await navigator.serviceWorker.register('./service-worker.js');
      await navigator.serviceWorker.ready;

      // Ask the SW to cache every component needed by PDF and spreadsheets.
      const controller=navigator.serviceWorker.controller || reg.active || reg.waiting;
      if(controller){
        controller.postMessage({type:'PREPARE_OFFLINE'});
      }

      // Confirm from cache state.
      const check=async()=>{
        try{
          const resp=await fetch('./offline-ready.json',{cache:'no-store'});
          if(resp.ok){
            setStatus(navigator.onLine?'✓ Pronto para uso offline':'✓ Funcionando offline',
                      navigator.onLine?'ready':'offline');
          }
        }catch{
          if(!navigator.onLine)setStatus('✓ Funcionando offline','offline');
        }
      };
      setTimeout(check,1800);
      setTimeout(check,4500);
    }catch(err){
      console.error(err);
      setStatus('Não foi possível preparar o modo offline','error');
    }
  });

  navigator.serviceWorker.addEventListener('message',event=>{
    if(event.data?.type==='OFFLINE_READY'){
      setStatus(navigator.onLine?'✓ Pronto para uso offline':'✓ Funcionando offline',
                navigator.onLine?'ready':'offline');
    }
    if(event.data?.type==='OFFLINE_ERROR'){
      setStatus('Falha ao baixar componentes offline','error');
    }
  });

  window.addEventListener('online',()=>setStatus('✓ Pronto para uso offline','ready'));
  window.addEventListener('offline',updateNetworkBadge);
  updateNetworkBadge();
})();
</script>
